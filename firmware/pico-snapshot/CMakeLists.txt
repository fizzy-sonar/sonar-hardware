cmake_minimum_required(VERSION 3.13)
include(pico_sdk_import.cmake)
project(sonar_pico_snapshot C CXX ASM)
pico_sdk_init()
add_executable(sonar_pico_snapshot src/main.c src/snapshot_protocol.c src/capture_core.c src/pdm_dma.c)
pico_generate_pio_header(sonar_pico_snapshot ${CMAKE_CURRENT_LIST_DIR}/pio/pdm_capture.pio)
target_include_directories(sonar_pico_snapshot PRIVATE include)
target_link_libraries(sonar_pico_snapshot pico_stdlib hardware_pio hardware_dma tinyusb_device tinyusb_board)
pico_enable_stdio_usb(sonar_pico_snapshot 1)
pico_enable_stdio_uart(sonar_pico_snapshot 0)
pico_add_extra_outputs(sonar_pico_snapshot)
